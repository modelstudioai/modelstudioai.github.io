## 怎么参赛

1. **选方向**：带着原创想法报名，或到阿宝许愿池找一个有共鸣的真实愿望把它做出来。
2. **做作品**：新手可用云端工具对话式生成应用；开发者可本地研发，用阿里云百炼 CLI、Agent Skills 调用模型与工具完成开发。
3. **提报作品**：本地研发作品按大赛要求经 QMuse 托管或支付宝开放平台上传；作品类型与上传方式以大赛官网最新公告为准。

document.addEventListener('DOMContentLoaded', function () {
  document.querySelectorAll('.b-copy').forEach(function (btn) {
    btn.addEventListener('click', function () {
      var url = btn.getAttribute('data-url');
      function done() {
        btn.textContent = '已复制 ✓';
        btn.classList.add('copied');
        setTimeout(function () { btn.textContent = '复制'; btn.classList.remove('copied'); }, 1600);
      }
      function fallback() {
        var ta = document.createElement('textarea');
        ta.value = url;
        ta.style.position = 'fixed';
        ta.style.opacity = '0';
        document.body.appendChild(ta);
        ta.select();
        try { document.execCommand('copy'); done(); } catch (e) {}
        document.body.removeChild(ta);
      }
      if (navigator.clipboard && navigator.clipboard.writeText) {
        navigator.clipboard.writeText(url).then(done, fallback);
      } else { fallback(); }
    });
  });
});
